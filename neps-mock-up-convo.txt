```python
@app.get("/api/participants")

    data = store.get_participants(country=country, site=site)

    """Get single participant"""
    participant = next(

    if not participant:
        raise HTTPException(404, "Participant not found")

    records = store.reports
```

### Client Code (Same for All Teams)

```python
# app/clients/redcap_client.py
import os
import requests
from typing import Optional, List

REDCAP_URL = os.getenv("REDCAP_API_URL", "https://neps-mock-api-production.up.railway.app/api")
REDCAP_TOKEN = os.getenv("REDCAP_API_TOKEN", "mock_token_neps_2025")

class RedCapClient:
    def __init__(self):
        self.base_url = REDCAP_URL.rstrip('/')
        self.headers = {"Authorization": f"Bearer {REDCAP_TOKEN}"}
    
    def get_participants(self, country: Optional[str] = None) -> list:
        params = {"country": country} if country else {}
        response = requests.get(f"{self.base_url}/participants", params=params)
        return response.json()["data"]
    
    def export_records(self) -> list:
        response = requests.get(f"{self.base_url}/records")
        return response.json()

# Usage — same code works for mock AND real REDCap
client = RedCapClient()
participants = client.get_participants(country="Ghana")
```
